## Р‘Р°Р·РѕРІР°СЏ СЃРёСЃС‚РµРјР° Р±Р»РёР¶РЅРµРіРѕ Р±РѕСЏ.
## Р›РљРњ вЂ” Р±С‹СЃС‚СЂР°СЏ Р°С‚Р°РєР°, РџРљРњ вЂ” С‚СЏР¶С‘Р»Р°СЏ Р°С‚Р°РєР°, Q вЂ” Р±Р»РѕРє.

@export var light_damage: float = 12.0
@export var heavy_damage: float = 28.0
@export var light_stamina_cost: float = 8.0
@export var heavy_stamina_cost: float = 20.0

@export var attack_radius: float = 1.0
@export var attack_cooldown: float = 0.38

var _player: CharacterBody3D
var _stats: Node
var _cooldown_remaining: float = 0.0
var _blocking: bool = false


func _ready() -> void:
    _player = get_parent() as CharacterBody3D

    if _player == null:
        push_error("CombatController РґРѕР»Р¶РµРЅ Р±С‹С‚СЊ РґРѕС‡РµСЂРЅРёРј СѓР·Р»РѕРј CharacterBody3D.")
        set_process_unhandled_input(false)
        set_physics_process(false)
        return

    _stats = _player.get_node_or_null("PlayerStats")

    if _stats == null:
        push_warning("CombatController: СѓР·РµР» PlayerStats РїРѕРєР° РЅРµ РЅР°Р№РґРµРЅ.")


func _physics_process(delta: float) -> void:
    _cooldown_remaining = maxf(0.0, _cooldown_remaining - delta)


func _unhandled_input(event: InputEvent) -> void:
    if _player == null:
        return

    if event is InputEventMouseButton:
        var mouse_event := event as InputEventMouseButton

        if not mouse_event.pressed:
            return

        if Input.mouse_mode != Input.MOUSE_MODE_CAPTURED:
            return

        match mouse_event.button_index:
            MOUSE_BUTTON_LEFT:
                _try_attack(false)
                get_viewport().set_input_as_handled()
            MOUSE_BUTTON_RIGHT:
                _try_attack(true)
                get_viewport().set_input_as_handled()

    elif event is InputEventKey:
        var key_event := event as InputEventKey

        if key_event.keycode == KEY_Q:
            _blocking = key_event.pressed and not key_event.echo
            get_viewport().set_input_as_handled()


func _try_attack(is_heavy: bool) -> void:
    if _cooldown_remaining > 0.0:
        return

    if _stats != null and _stats.has_method("is_alive"):
        if not bool(_stats.call("is_alive")):
            return

    var stamina_cost := heavy_stamina_cost if is_heavy else light_stamina_cost

    if _stats != null and _stats.has_method("try_consume_stamina"):
        var consumed: bool = bool(
            _stats.call("try_consume_stamina", stamina_cost)
        )

        if not consumed:
            print("[Р‘РћР™] РќРµРґРѕСЃС‚Р°С‚РѕС‡РЅРѕ РІС‹РЅРѕСЃР»РёРІРѕСЃС‚Рё.")
            return

    _cooldown_remaining = attack_cooldown

    var damage := heavy_damage if is_heavy else light_damage
    var attack_name := "С‚СЏР¶С‘Р»Р°СЏ" if is_heavy else "Р±С‹СЃС‚СЂР°СЏ"

    print("[Р‘РћР™] ", attack_name, " Р°С‚Р°РєР°. РЈСЂРѕРЅ: ", damage)
    _perform_hit(damage)


func _perform_hit(damage: float) -> void:
    var space_state := _player.get_world_3d().direct_space_state
    var forward := -_player.global_transform.basis.z.normalized()
    var hit_center := (
        _player.global_position
        + Vector3.UP
        + forward * attack_range * 0.65
    )

    var sphere := SphereShape3D.new()
    sphere.radius = attack_radius

    var query := PhysicsShapeQueryParameters3D.new()
    query.shape = sphere
    query.transform = Transform3D(Basis.IDENTITY, hit_center)
    query.collision_mask = 0xFFFFFFFF
    query.exclude = [_player.get_rid()]
    query.collide_with_bodies = true
    query.collide_with_areas = true

    var results := space_state.intersect_shape(query, 32)
    var damaged_targets: Array[Node] = []

    for result in results:
        var collider: Object = result.get("collider")

        if not collider is Node:
            continue

        var target := _find_enemy_root(collider as Node)

        if target == null or damaged_targets.has(target):
            continue

        if target.has_method("apply_damage"):
            damaged_targets.append(target)
            target.call("apply_damage", damage)


func _find_enemy_root(node: Node) -> Node:
    var current: Node = node

    while current != null and current != _player:
        if current.is_in_group("enemies"):
            return current
        current = current.get_parent()

    return null


func is_blocking() -> bool:
    return Input.is_key_pressed(KEY_Q)

func get_incoming_damage_multiplier() -> float:
    return 0.35 if is_blocking() else 1.0

func _physics_process(delta: float) -> void:
    if is_dead:
        return

    attack_timer = maxf(0.0, attack_timer - delta)

    if not is_instance_valid(player):
        player = _find_player()

    if not is_instance_valid(player):
        _apply_gravity(delta)
        velocity.x = move_toward(velocity.x, 0.0, move_speed * 4.0 * delta)
        velocity.z = move_toward(velocity.z, 0.0, move_speed * 4.0 * delta)
        move_and_slide()
        return

    var offset := player.global_position - global_position
    var flat_offset := Vector3(offset.x, 0.0, offset.z)
    var distance := flat_offset.length()

    if distance > detection_range:
        velocity.x = move_toward(velocity.x, 0.0, move_speed * 4.0 * delta)
        velocity.z = move_toward(velocity.z, 0.0, move_speed * 4.0 * delta)
        _apply_gravity(delta)
        move_and_slide()
        return

    if distance > 0.05:
        look_at(
            Vector3(player.global_position.x, global_position.y, player.global_position.z),
            Vector3.UP
        )

    if distance > attack_range:
        var direction := flat_offset.normalized()
        velocity.x = direction.x * move_speed
        velocity.z = direction.z * move_speed
    else:
        velocity.x = move_toward(velocity.x, 0.0, move_speed * 6.0 * delta)
        velocity.z = move_toward(velocity.z, 0.0, move_speed * 6.0 * delta)
        if attack_timer <= 0.0:
            _attack_player()

    _apply_gravity(delta)
    move_and_slide()


func _find_player() -> Node3D:
    var candidate := get_tree().get_first_node_in_group("player")
    if candidate is Node3D and candidate != self:
        return candidate as Node3D
    return null


func _apply_gravity(delta: float) -> void:
    if not is_on_floor():
        velocity.y -= gravity * delta
    else:
        velocity.y = 0.0


func _attack_player() -> void:
    attack_timer = attack_cooldown

    if not is_instance_valid(player):
        return

    var multiplier := 1.0
    var controller := player.find_child("CombatController", true, false)
    if controller != null and controller.has_method("get_incoming_damage_multiplier"):
        multiplier = float(controller.call("get_incoming_damage_multiplier"))

    var final_damage := attack_damage * clampf(multiplier, 0.0, 1.0)
    var stats := player.find_child("PlayerStats", true, false)

    if stats == null:
        stats = player.find_child("PlayerStats", false, false)

    if stats != null:
        if stats.has_method("take_damage"):
            stats.call("take_damage", final_damage)
        elif stats.has_method("apply_damage"):
            stats.call("apply_damage", final_damage)
        elif stats.has_method("receive_damage"):
            stats.call("receive_damage", final_damage)
        else:
            push_warning("Enemy hit player, but PlayerStats has no supported damage method.")
    else:
        push_warning("Enemy hit player, but PlayerStats node was not found.")

    print("Enemy attack: ", final_damage, " damage (block multiplier ", multiplier, ")")


func apply_damage(amount: float) -> void:
    if is_dead or amount <= 0.0:

    health = maxf(0.0, health - amount)
    _update_health_label()
    print("Enemy AI HP: ", health, "/", max_health)

    if health <= 0.0:
        _die()

func try_consume_stamina(amount: float) -> bool:
    if dead or amount <= 0.0:
        sprinting = false
        return false

    if stamina < amount:
        sprinting = false
        return false

    stamina = maxf(0.0, stamina - amount)
    sprinting = true
    stamina_changed.emit(stamina, max_stamina)
    _refresh_hud()
    return true

    if dead or amount <= 0.0:
